        internal FontWithColor ShowWindow(FontWithColor old_font)
        {
            old = old_font;
            ShowDialog();
            if (accept)
            {
                double left, bottom;
                if (!double.TryParse(bottom_textbox.Text, out bottom) || bottom > 250 || bottom < 10)
                    bottom = old.position.Y;
                if (!double.TryParse(left_textbox.Text, out left) || left > 200 || left < 10)
                    bottom = old.position.X;
                return new FontWithColor(FontsName.SelectedIndex, FontColors.SelectedIndex, FontSizes.Value, Italic.IsChecked.Value, Bold.IsChecked.Value, Encoding.GetEncoding(Convert.ToInt32(((ComboBoxItem)encode_comboBox.SelectedItem).Tag)), (sync_multiple.SelectedIndex * 59 + 1) * sync_slider.Value * 1000, left, bottom, background_colors_list.SelectedIndex, text_opacity_slider.Value, background_opacity_slider.Value);
            }
            else
                return old;
        }

        private void Cancel(object sender, RoutedEventArgs e)

        }

        private void Window_Loaded(object sender, RoutedEventArgs e)
        {
            statring_window = false;
            ObservableCollection<ColorItem> color_data = new ObservableCollection<ColorItem>();
            background_colors_list.DataContext = FontColors.DataContext = color_data;
            ObservableCollection<FontItem> font_data = new ObservableCollection<FontItem>();
            FontsName.DataContext = font_data;
            rec.Width = label1.ActualWidth;
            foreach (var i in System.Drawing.FontFamily.Families)
                font_data.Add(new FontItem(new FontFamily(i.Name), i.Name));
            for (int i = 0; i < FontWithColor.BrushesNames.Length; i++)
                color_data.Add(new ColorItem((SolidColorBrush)FontWithColor.WMBrushes[i], FontWithColor.BrushesNames[i]));
            FontColors.Items.Refresh();
            FontsName.Items.Refresh();
            foreach (var i in Encoding.GetEncodings())
            {
                ComboBoxItem item = new ComboBoxItem();
                item.Content = i.DisplayName;
                item.Tag = i.CodePage;
                encode_comboBox.Items.Add(item);
            }
            if (old != null)
            {
                for (int i = 0; i < font_data.Count; i++)
                    if (old.font_name.Name.ToLower().Equals(font_data[i].FontName.ToLower()))
                    {
                        FontsName.SelectedIndex = i;
                        break;
                    }
                if (FontsName.SelectedIndex == -1)
                    FontsName.SelectedIndex = 0;
                else
                    FontsName.ScrollIntoView(FontsName.Items[FontsName.SelectedIndex]);
                if(FontColors.SelectedIndex>-1)
                {
                    FontColors.SelectedIndex = FontWithColor.GetColorIndex(old.font_color);
                    FontColors.ScrollIntoView(FontColors.Items[FontColors.SelectedIndex]);
                }
                FontSizes.Value = old.font_size;
                TextTest.FontSize = FontSizes.Value;
                Bold.IsChecked = old.bold;
                Italic.IsChecked = old.italic;
                if (old.async_millisecond <= 60000 && old.async_millisecond >= -60000)
                    sync_slider.Value = old.async_millisecond / 1000;
                else
                {
                    sync_slider.Value = old.async_millisecond / 60000;
                    sync_multiple.SelectedIndex = 1;
                }
                for (int i = 0; i < encode_comboBox.Items.Count; i++)
                    if (Convert.ToInt32(((ComboBoxItem)encode_comboBox.Items[i]).Tag) == old.text_encode.CodePage)
                    {
                        encode_comboBox.SelectedIndex = i;
                        break;
                    }
                if (encode_comboBox.SelectedIndex == -1)
                    encode_comboBox.SelectedIndex = 0;
                left_textbox.Text = old.position.X.ToString();
                bottom_textbox.Text = old.position.Y.ToString();
            }
        }
